基本数据类型

Python 2025-10-16 6415 字 约 17 分钟
python

在python中除int,float,char三中类型之外，还有字节(bytes)，元组(tuple)，列表(list),集合(set)和字典(dict)等类型

元组(不可变，有序)

t1 = (1, 2, 3)          # 标准创建
t2 = 4, 5, 6            # 括号可省略
single = (7,)           # 单元素元组（必须加逗号）
print(t1[0])            # 1 → 索引访问
print(t1[1:])           # (2, 3) → 切片
a, b, c = t1            # 解包赋值
new_t = t1 + (4,)       # (1, 2, 3, 4) → 连接
rep_t = t1 * 2          # (1, 2, 3, 1, 2, 3) → 重复
t1.index(2)             # 1 → 元素索引，输入元素，返回第一个索引
t1.count(3)             # 1 → 元素计数，输入元素，返回元组中该元素个数
len(t1)                 # 3 → 长度

列表(可变，有序)

lst = [1, "a", True]    # 混合类型
empty = []              # 空列表
nested = [[1,2], [3,4]] # 嵌套列表
lst.append(4)           # [1, "a", True, 4]
lst.insert(1, "b")      # [1, "b", "a", True, 4]
lst[0] = 100            # 修改元素
del lst[2]              # 删除索引2元素
lst.pop()               # 移除并返回末尾元素
lst.remove("a")         # 移除首个匹配元素
print(lst[1])           # "b" → 索引访问
print(lst[-1])          # 4 → 负索引
print("b" in lst)       # True → 成员检查
lst.sort()              # 排序（同类型元素才能排序，否则报错）
sorted_lst = sorted(lst) # 返回新排序列表
lst[1:3] = [10, 20]     # 切片赋值
lst.reverse()           # 反转列表
lst.copy()              # 浅拷贝

集合(可变，无序，不可重复)

s1 = {1, 2, 3}          # 字面量创建
s2 = set([1, 2, 2, 3])  # {1, 2, 3} → 从列表转换（自动去重）
empty_set = set()       # 空集合（不能用{}，这是字典）
s1.add(4)               # {1, 2, 3, 4}
s1.update([5, 6])       # {1, 2, 3, 4, 5, 6}
s1.remove(3)            # 移除元素（不存在报错）
s1.discard(10)          # 安全移除（不存在不报错）
s1.pop()                # 随机移除一个元素
A = {1, 2, 3}; B = {3, 4, 5}
A | B   # {1,2,3,4,5} → 并集
A & B   # {3} → 交集
A - B   # {1,2} → 差集
A ^ B   # {1,2,4,5} → 对称差集
A <= B  # False → 子集检查
A.isdisjoint(B) # False → 无交集检查

字典(可变，键唯一，3.7+ 保持插入顺序)

d = {"name": "Tom", "age": 18}   # 字面量创建
d2 = dict(a=1, b=2)              # 关键字创建
empty = {}                       # 空字典（注意 {} 是字典不是集合）
d["city"] = "BJ"                 # 新增/修改（键存在则覆盖）
d.get("age")                     # 18 → 取值（键不存在返回 None）
d.get("x", 0)                    # 0 → 键不存在时返回默认值
del d["age"]                     # 删除键
d.pop("city")                    # 删除并返回对应值
d.keys()                         # 所有键 dict_keys(['name'])
d.values()                       # 所有值
d.items()                        # 键值对 dict_items([('name', 'Tom')])
"name" in d                      # True → 成员检查（查的是键）
d.update({"age": 20})            # 批量更新/合并另一个字典
d.setdefault("score", 60)        # 键存在返回其值；不存在则插入默认值并返回
len(d)                           # 键值对个数
d.clear()                        # 清空
for k in d: ...                  # 遍历键
for k, v in d.items(): ...       # 遍历键值对（最常用）

字符串(不可变，有序)

s = "Hello Python"
s.upper()                        # 转大写
s.lower()                        # 转小写
s.strip()                        # 去两端空白（lstrip 左 / rstrip 右）
s.split(",")                     # 按分隔符切成列表
"-".join(["a", "b"])             # 列表拼成字符串 → "a-b"
s.replace("a", "b")              # 替换（返回新串）
s.find("Py")                     # 子串索引，找不到返回 -1
s.index("Py")                    # 同 find，找不到报错
s.startswith("He")               # 前缀判断（endswith 后缀）
s.isdigit()                      # 是否全为数字（isalpha 全字母）
s.zfill(5)                       # 左补 0 到指定长度
len(s)                           # 长度
s[::-1]                          # 反转
f"{s}!"                          # f-string 格式化

字节 bytes(不可变) / bytearray(可变)

b = b"abc"                       # bytes 字面量
ba = bytearray(b"abc")           # bytearray，元素可改
b = "中文".encode("utf-8")        # str → bytes
s = b.decode("utf-8")            # bytes → str
b.hex()                          # 转十六进制字符串
bytes.fromhex("6162")            # 十六进制 → b'ab'
list(b)                          # [97, 98, 99]
b[0]                             # 97（取到的是 int，不是字符）

str/bytes 编码细节见 Python3-str与bytes编码.md。

数据转换（类型之间 + 进制之间）

场景：解题时数据在 int / str / bytes / list / hex 串之间来回倒；以及"十进制转 8 位二进制"这类格式需求。
下面每条的输出都在本机 py 实际跑过。

1. 数值类型之间

int(3.9)        # 3      浮点→整数：直接丢掉小数，不是四舍五入
int(-3.9)       # -3     向零截断
round(3.5)      # 4      四舍五入？不完全是
round(2.5)      # 2      因为 round 是"银行家舍入"（.5 取偶）：2.5→2、3.5→4
round(3.14159, 2)  # 3.14  保留两位
float(3)        # 3.0    整数→浮点
abs(-5)         # 5      取绝对值
bool(0)         # False  0/0.0/""/[]/{}/None → False，其余 True
bool(3)         # True

坑：想四舍五入别用 int()。真要四舍五入用 int(x + 0.5)（正数）或 round()（但它按 .5 取偶）。

2. 数值 ↔ 字符串

str(42)          # '42'
int("42")        # 42     字符串→整数
int("  42  ")    # 42     两端空白会被自动忽略
float("3.14")    # 3.14
int("3.14")      # ❌ ValueError：int() 不吃小数点和指数
int(float("3.14"))  # 3    想转小数串就先 float 再 int
str(3.14000)     # '3.14' 转成字符串后多余 0 就没了

3. 字符串 ↔ 列表（字符 / ASCII 码）

list("abc")                        # ['a', 'b', 'c']
"".join(['a', 'b', 'c'])           # 'abc'     列表拼回字符串
"1 2 3".split()                    # ['1', '2', '3']  按空白拆（空格/换行/制表符都行）
"a,b,c".split(",")                 # ['a', 'b', 'c']  按指定分隔符拆
list(map(int, "1 2 3".split()))    # [1, 2, 3]        拆完顺手转 int

''.join(chr(n) for n in [102, 108, 97, 103])   # 'flag'  数字列表 → 字符串
[ord(c) for c in "flag"]                       # [102, 108, 97, 103] 字符串 → 数字列表

CTF 高频：题目给 83 89 78 84 这种串 → list(map(int, s.split())) → chr 变字符。

4. 容器之间互转

list((1, 2, 3))          # [1, 2, 3]          元组→列表
tuple([1, 2, 3])         # (1, 2, 3)          列表→元组
set([1, 1, 2, 3])        # {1, 2, 3}          转集合顺手去重
sorted(set([3, 1, 2]))   # [1, 2, 3]          集合无序，要顺序就 sorted
list({"a": 1}.keys())    # ['a']              字典键→列表
list({"a": 1}.items())   # [('a', 1)]         键值对→列表
dict([("a", 1), ("b", 2)])   # {'a': 1, 'b': 2}  键值对列表→字典
dict(zip("ab", [1, 2]))      # {'a': 1, 'b': 2}  zip 成对再转字典

5. 字符串 ↔ bytes（编码）

"flag".encode()          # b'flag'    str → bytes
b"flag".decode()         # 'flag'     bytes → str
"中文".encode()          # b'\xe4\xb8\xad\xe6\x96\x87'  UTF-8 下一个汉字 3 字节
len("中文".encode())     # 6

b"flag".hex()            # '666c6167' bytes → 十六进制串
bytes.fromhex("666c6167")     # b'flag'  十六进制串 → bytes
list(b"abc")             # [97, 98, 99]   bytes→整数列表（每个元素本来就是 0-255 的 int）
bytes([97, 98, 99])      # b'abc'        整数列表→bytes
b"abc"[0]                # 97            取出来是 int，不是字符

整数 ↔ bytes（RSA 解出大整数要变明文就靠它）：

n = 123456789
n.to_bytes(4, "big")           # b'\x07[\xcd\x15'  整数→bytes，指定字节数
n.to_bytes(4, "big").hex()     # '075bcd15'
int.from_bytes(b"\x07[\xcd\x15", "big")   # 123456789  bytes→整数

大整数不知道长度时：(n.bit_length() + 7) // 8 算字节数。"big" 是大端（高位在前），CTF 里几乎都用它。

6. 进制转换（重点）

6.1 十进制 → 二/八/十六进制

三种写法，看你要不要前缀：

bin(10)      # '0b1010'   带前缀
oct(10)      # '0o12'
hex(255)     # '0xff'

format(255, "b")   # '11111111'  不带前缀
format(255, "x")   # 'ff'
format(255, "#x")  # '0xff'      想要前缀就加 #
f"{255:x}"         # 'ff'        f-string 里同样可用

6.2 按固定宽度输出（为什么要 8 位二进制）

format(5, "08b")     # '00000101'   8 位二进制，左侧补 0
format(5, "02x")     # '05'         2 位十六进制
bin(5)[2:].zfill(8)  # '00000101'   不用 format 的等价写法
format(-1 & 0xFF, "08b")   # '11111111'   负数要按补码就与 0xFF 掩一下

格式说明符 08b 拆开看：0 = 补 0，8 = 总宽度（不足补齐、超了原样输出），b/x/o = 二/十六/八进制。
注意不做截断：format(256, "08b") 得到 9 位 '100000000'，要严格 8 位先 n & 0xFF。

6.3 任意进制 ↔ 十进制

int("1010", 2)    # 10     二进制串 → 十进制
int("777", 8)     # 511    八进制串 → 十进制
int("ff", 16)     # 255    十六进制串 → 十进制
int("0x1f", 16)   # 31     带前缀也能认（base 写 16 时 0x 可留可去）
int("10", 36)     # 36     base 最大 36，用 0-9 a-z 表示，能当"字符串哈希"用

反过来（十进制 → 任意进制）没有内置函数，手写五行：

def to_base(n, base):
    if n == 0:
        return "0"
    digits = "0123456789abcdefghijklmnopqrstuvwxyz"
    sign = "-" if n < 0 else ""
    n, out = abs(n), ""
    while n:
        out = digits[n % base] + out   # 取余得当前位，除基继续
        n //= base
    return sign + out

to_base(255, 16)   # 'ff'
to_base(255, 2)    # '11111111'
to_base(1000, 7)   # '2626'      验证：int('2626', 7) == 1000

二进制 ↔ 十六进制的关系：1 位十六进制 = 4 位二进制。0xf = 1111，所以 0xff = 1111 1111。
反过来换算：hex(int("1010", 2)) → '0xa'（先转十进制再转十六进制）。

7. 容易踩的坑

  1. 转换函数只认整数，不认浮点：bin(3.0)、hex(3.0) 都报 TypeError，得先 int()。
  2. 负数不会被当补码：bin(-5) → '-0b101'（带负号），要 8 位补码用 format(-5 & 0xFF, "08b")。
  3. bytes 和 str 之间没有隐式转换：str(b"abc") 得到 "b'abc'"（连 b 和引号一起），要 b"abc".decode()。int(b"42") 倒是能工作。
  4. format(n, "08b") 不是"截断到 8 位"，是"至少 8 位"。想截断用 n & 0xFF。
  5. 集合转列表顺序不固定，要稳定顺序一律 sorted(set(...))。

8. 一图串起常见解题链路

# hex 串 → 整数 → 字节 → 明文（RSA 结尾常用）
h = "68656c6c6f"
m = int(h, 16)                    # 十进制整数
m.to_bytes((m.bit_length() + 7) // 8, "big").decode()   # 'hello'

# 数字串 → 字符（凯撒/ASCII 题）
list(map(int, "102 108 97 103".split()))   # [102, 108, 97, 103]
''.join(chr(x) for x in [102, 108, 97, 103])   # 'flag'

# 十进制 → 8 位二进制（掩码/位运算题）
format(0x5A, "08b")   # '01011010'
